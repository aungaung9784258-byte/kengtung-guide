/// ---------------------------------------------------------------------------
/// Hero gallery used at the top of the place-detail screen.
///
///  * big photo = swipeable PageView
///  * thumbnail strip underneath = tap to jump, the active one is highlighted
///  * tap the big photo (or the expand chip) = full-screen viewer with
///    two-finger pinch zoom
/// ---------------------------------------------------------------------------
class PlaceGallery extends StatefulWidget {
  const PlaceGallery({
    super.key,
    required this.place,
    this.heroTag,
  });

  final Place place;
  final String? heroTag;

  @override
  State<PlaceGallery> createState() => _PlaceGalleryState();
}

class _PlaceGalleryState extends State<PlaceGallery> {
  late final PageController _page = PageController();
  final _thumbs = ScrollController();
  int _index = 0;

  int get _count =>
      widget.place.images.isEmpty ? 1 : widget.place.images.length;

  @override
  void dispose() {
    _page.dispose();
    _thumbs.dispose();
    super.dispose();
  }

  void _goTo(int i) {
    _page.animateToPage(
      i,
      duration: AppMotion.base,
      curve: AppMotion.easeOutExpoCurve,
    );
  }

  void _onPage(int i) {
    setState(() => _index = i);
    // Keep the active thumbnail in view.
    if (_thumbs.hasClients) {
      const itemExtent = 72.0 + 10.0; // width + separator
      final target = (i * itemExtent) - 90;
      _thumbs.animateTo(
        target.clamp(0.0, _thumbs.position.maxScrollExtent),
        duration: AppMotion.base,
        curve: Curves.easeOut,
      );
    }
  }

  void _openFullscreen() {
    Navigator.of(context).push(
      PageRouteBuilder(
        opaque: false,
        barrierColor: Colors.black,
        transitionDuration: AppMotion.base,
        pageBuilder: (_, a, __) => FadeTransition(
          opacity: a,
          child: FullscreenGallery(
            place: widget.place,
            initialIndex: _index,
          ),
        ),
      ),
    ).then((returned) {
      // Keep the hero in sync with wherever the user ended up.
      if (returned is int && returned != _index && mounted) {
        _page.jumpToPage(returned);
        _onPage(returned);
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    final showThumbs = _count > 1;

    Widget pager = PageView.builder(
      controller: _page,
      onPageChanged: _onPage,
      physics: const BouncingScrollPhysics(),
      itemCount: _count,
      itemBuilder: (_, i) => GestureDetector(
        onTap: _openFullscreen,
        child: PlaceImage(
          place: widget.place,
          index: i,
          decodeWidth: 1080,
        ),
      ),
    );

    // Only the first photo participates in the shared-element transition,
    // otherwise Flutter sees duplicate tags mid-swipe.
    if (widget.heroTag != null && _index == 0) {
      pager = Hero(tag: widget.heroTag!, child: pager);
    }

    return Stack(
      fit: StackFit.expand,
      children: [
        pager,

        const Positioned.fill(
          child: IgnorePointer(
            child: DecoratedBox(
              decoration: BoxDecoration(gradient: AppColors.heroScrim),
            ),
          ),
        ),

        // counter + "tap to zoom" affordance
        Positioned(
          right: 20,
          bottom: showThumbs ? 134 : 74,
          child: GestureDetector(
            onTap: _openFullscreen,
            child: Container(
              padding:
                  const EdgeInsets.symmetric(horizontal: 11, vertical: 7),
              decoration: BoxDecoration(
                color: Colors.black.withValues(alpha: 0.45),
                borderRadius: BorderRadius.circular(20),
              ),
              child: Row(
                mainAxisSize: MainAxisSize.min,
                children: [
                  const Icon(Icons.zoom_out_map_rounded,
                      size: 13, color: Colors.white),
                  if (_count > 1) ...[
                    const SizedBox(width: 7),
                    Text(
                      '${_index + 1}/$_count',
                      style: const TextStyle(
                        color: Colors.white,
                        fontSize: 11.5,
                        fontWeight: FontWeight.w700,
                      ),
                    ),
                  ],
                ],
              ),
            ),
          ),
        ),

        // thumbnail strip
        if (showThumbs)
          Positioned(
            left: 20,
            right: 20,
            bottom: 66,
            child: SizedBox(
              height: 62,
              child: ListView.separated(
                controller: _thumbs,
                scrollDirection: Axis.horizontal,
                physics: const BouncingScrollPhysics(),
                itemCount: _count,
                separatorBuilder: (_, __) => const SizedBox(width: 10),
                itemBuilder: (_, i) {
                  final active = i == _index;
                  return RevealIn(
                    index: i,
                    scaleFrom: 0.8,
                    child: GestureDetector(
                      onTap: () => _goTo(i),
                      child: AnimatedContainer(

/// Full-screen photo viewer: swipe between photos, pinch with two fingers to

class FullscreenGallery extends StatefulWidget {
  const FullscreenGallery({
    super.key,
    required this.place,
    this.initialIndex = 0,
  });

  final Place place;
  final int initialIndex;

  @override
  State<FullscreenGallery> createState() => _FullscreenGalleryState();
}

class _FullscreenGalleryState extends State<FullscreenGallery>
    with SingleTickerProviderStateMixin {
  late final PageController _page =
      PageController(initialPage: widget.initialIndex);
  late int _index = widget.initialIndex;

  /// One controller per page so each photo remembers its own zoom.
  final _zoom = <int, TransformationController>{};

  late final AnimationController _anim = AnimationController(
    vsync: this,
    duration: AppMotion.base,
  );
  Animation<Matrix4>? _zoomAnim;

  bool get _isZoomed {
    final c = _zoom[_index];
    return c != null && c.value.getMaxScaleOnAxis() > 1.02;
  }

  TransformationController _ctrl(int i) =>
      _zoom.putIfAbsent(i, () => TransformationController());

  @override
  void dispose() {
    _anim.dispose();
    for (final c in _zoom.values) {
      c.dispose();
    }
    _page.dispose();
    super.dispose();
  }

  void _animateTo(Matrix4 target) {
    final c = _ctrl(_index);
    _zoomAnim = Matrix4Tween(begin: c.value, end: target)
        .animate(CurvedAnimation(parent: _anim, curve: AppMotion.emphasized));
    void listener() => c.value = _zoomAnim!.value;
    _anim
      ..removeListener(listener)
      ..addListener(listener)
      ..forward(from: 0);
  }

  void _handleDoubleTap(TapDownDetails d, int i) {
    final c = _ctrl(i);
    if (c.value.getMaxScaleOnAxis() > 1.02) {
      _animateTo(Matrix4.identity());
    } else {
      const scale = 2.5;
      final p = d.localPosition;
      // Built with setEntry instead of translate()/scale(): those two are
      // deprecated in current vector_math, and this is the same matrix.
      _animateTo(
        Matrix4.identity()
          ..setEntry(0, 0, scale)
          ..setEntry(1, 1, scale)
          ..setEntry(0, 3, -p.dx * (scale - 1))
          ..setEntry(1, 3, -p.dy * (scale - 1)),
      );
    }
    setState(() {});
  }

  @override
  Widget build(BuildContext context) {
    final count =
        widget.place.images.isEmpty ? 1 : widget.place.images.length;

    return Scaffold(
      backgroundColor: Colors.black,
      body: Stack(
        children: [
          PageView.builder(
            controller: _page,
            itemCount: count,
            // While a photo is zoomed in, dragging should pan it, not flick
            // to the next photo.
            physics: _isZoomed
                ? const NeverScrollableScrollPhysics()
                : const BouncingScrollPhysics(),
            onPageChanged: (i) => setState(() => _index = i),
            itemBuilder: (_, i) {
              final c = _ctrl(i);
              return GestureDetector(
                onDoubleTapDown: (d) => _handleDoubleTap(d, i),
                onDoubleTap: () {},
                child: InteractiveViewer(
                  transformationController: c,
                  minScale: 1,
                  maxScale: 5,
                  // Let the zoomed photo be dragged past its own edges a
                  // little; it feels much less rigid.
                  boundaryMargin: const EdgeInsets.all(40),
                  onInteractionEnd: (_) => setState(() {}),
                  child: Center(
                    child: PlaceImage(
                      place: widget.place,
                      index: i,
                      fit: BoxFit.contain,
                      decodeWidth: 1600,
                    ),
                  ),
                ),
              );
            },
          ),

          // close
          Positioned(
            top: MediaQuery.of(context).padding.top + 10,
            left: 14,
            child: _GlassIcon(
              icon: Icons.close_rounded,
              onTap: () => Navigator.of(context).pop(_index),
            ),
          ),

          // counter
          if (count > 1)
            Positioned(
              top: MediaQuery.of(context).padding.top + 18,
              left: 0,
              right: 0,
              child: Center(
                child: Container(
                  padding: const EdgeInsets.symmetric(
                      horizontal: 14, vertical: 7),
                  decoration: BoxDecoration(
                    color: Colors.white.withValues(alpha: 0.14),
                    borderRadius: BorderRadius.circular(20),
                  ),
                  child: Text(
                    '${_index + 1} / $count',
                    style: const TextStyle(
                      color: Colors.white,
                      fontSize: 13,
                      fontWeight: FontWeight.w700,
                    ),
                  ),
                ),
              ),
            ),

          // page dots
          if (count > 1)
            Positioned(
              bottom: MediaQuery.of(context).padding.bottom + 26,
              left: 0,
              right: 0,
              child: Row(
                mainAxisAlignment: MainAxisAlignment.center,

/// Geometry helpers (equirectangular approximation — accurate well under 1 %
/// at town scale, and ~100x cheaper than haversine in a per-frame loop).

class NavGeo {
  static const double _metersPerDegLat = 111320.0;

  static double _mPerDegLng(double lat) =>
      _metersPerDegLat * math.cos(lat * math.pi / 180);

  /// Flat-earth metre offsets of [p] relative to [origin].
  static (double, double) _xy(LatLng p, LatLng origin) => (
        (p.longitude - origin.longitude) * _mPerDegLng(origin.latitude),
        (p.latitude - origin.latitude) * _metersPerDegLat,
      );

  static double distance(LatLng a, LatLng b) {
    final (dx, dy) = _xy(b, a);
    return math.sqrt(dx * dx + dy * dy);
  }

  /// Initial bearing from [a] to [b] in degrees, 0 = north, clockwise.
  static double bearing(LatLng a, LatLng b) {
    final (dx, dy) = _xy(b, a);
    final deg = math.atan2(dx, dy) * 180 / math.pi;
    return (deg + 360) % 360;
  }

  /// Perpendicular distance in metres from [p] to the segment a–b, plus the
  /// fraction (0..1) along the segment where the foot of the perpendicular is.
  static (double, double) distanceToSegment(LatLng p, LatLng a, LatLng b) {
    final (ax, ay) = (0.0, 0.0);
    final (bx, by) = _xy(b, a);
    final (px, py) = _xy(p, a);

    final dx = bx - ax;
    final dy = by - ay;
    final len2 = dx * dx + dy * dy;
    if (len2 == 0) return (math.sqrt(px * px + py * py), 0);

    var t = ((px - ax) * dx + (py - ay) * dy) / len2;
    t = t.clamp(0.0, 1.0);

    final cx = ax + t * dx;
    final cy = ay + t * dy;
    final ex = px - cx;
    final ey = py - cy;
    return (math.sqrt(ex * ex + ey * ey), t);
  }

  /// Interpolate between two points by fraction [t].
  static LatLng lerp(LatLng a, LatLng b, double t) => LatLng(
        a.latitude + (b.latitude - a.latitude) * t,
        a.longitude + (b.longitude - a.longitude) * t,
      );

  /// Shortest signed difference between two bearings, in (-180, 180].
  static double angleDelta(double from, double to) {
    var d = (to - from) % 360;
    if (d > 180) d -= 360;
    if (d <= -180) d += 360;
    return d;
  }

  /// Rotate [current] towards [target] by at most [maxStep] degrees.
  /// Keeps the compass from snapping when GPS heading is noisy.
  static double smoothAngle(double current, double target, double factor) =>
      current + angleDelta(current, target) * factor;
}

/// Where the user is relative to the route right now.
class RouteProgress {
  const RouteProgress({
    required this.snapped,
    required this.segmentIndex,
    required this.offRouteMeters,
    required this.remainingMeters,
    required this.courseBearing,
  });

  /// The user's position projected onto the route line.
  final LatLng snapped;
  final int segmentIndex;

  /// Perpendicular distance from the route line.
  final double offRouteMeters;
  final double remainingMeters;

  /// Direction the route is heading at this point (degrees, 0 = north).
  final double courseBearing;
}

/// ---------------------------------------------------------------------------
/// Live turn-by-turn-ish navigation session.
///
/// Google-Maps-like behaviour, built only from free pieces:
///   * follows the GPS stream and keeps the blue arrow on the user
///   * snaps the user onto the route line and counts down the remaining
///     distance / time
///   * detects when the user leaves the route and automatically fetches a new
///     one (throttled, so a bad GPS fix can't spam the router)
///   * reports arrival
///
/// It is a plain [ChangeNotifier] so the map screen can listen without
/// rebuilding anything else.
/// ---------------------------------------------------------------------------
class NavSession extends ChangeNotifier {
  NavSession({
    required this.destination,
    required this.mode,
    required RouteResult initialRoute,
    required this.isOnline,
    this.offRouteThresholdMeters = 45,
    this.arriveThresholdMeters = 35,
  }) : _route = initialRoute;

  final LatLng destination;
  TravelMode mode;

  /// Called by the owner so the session knows whether the router is reachable.
  final bool Function() isOnline;

  final double offRouteThresholdMeters;
  final double arriveThresholdMeters;

  RouteResult _route;
  RouteResult get route => _route;

  StreamSubscription<Position>? _sub;

  LatLng? _me;
  LatLng? get me => _me;

  /// Smoothed heading used to rotate the arrow and the camera.
  double _bearing = 0;
  double get bearing => _bearing;

  double _speedMps = 0;
  double get speedMps => _speedMps;

  RouteProgress? _progress;
  RouteProgress? get progress => _progress;

  bool _rerouting = false;
  bool get rerouting => _rerouting;

  bool _arrived = false;
  bool get arrived => _arrived;

  int _rerouteCount = 0;
  int get rerouteCount => _rerouteCount;

  /// Consecutive fixes that were off the line. Requiring a few in a row stops
  /// one noisy GPS reading from triggering a pointless reroute.
  int _offRouteStrikes = 0;
  DateTime _lastReroute = DateTime.fromMillisecondsSinceEpoch(0);

  /// Set false as soon as the user pans the map themselves.
  bool followCamera = true;

  /// Course-up (rotate the map with the user) vs north-up.
  bool courseUp = true;

  double get remainingMeters =>
      _progress?.remainingMeters ?? _route.distanceMeters;

  /// Remaining time, recomputed from the remaining distance so the ETA
  /// actually counts down as you drive.
  double get remainingSeconds {
    final total = _route.distanceMeters;
    if (total <= 0) return 0;
    return _route.durationSeconds * (remainingMeters / total);
  }

  String get remainingDistanceLabel => remainingMeters < 1000
      ? '${remainingMeters.round()} m'
      : '${(remainingMeters / 1000).toStringAsFixed(1)} km';

  String get remainingTimeLabel {
    final m = (remainingSeconds / 60).round();
    if (m < 1) return '<1 min';
    if (m < 60) return '$m min';
    return '${m ~/ 60} h ${m % 60} min';
  }

  /// Fraction of the route already travelled, 0..1 — drives the progress bar.
  double get completion {
    final total = _route.distanceMeters;
    if (total <= 0) return 0;
    return (1 - remainingMeters / total).clamp(0.0, 1.0);
  }

  // -------------------------------------------------------------------------

  void start({LatLng? initialPosition}) {
    if (initialPosition != null) _ingest(initialPosition, null, null);

    _sub = Geolocator.getPositionStream(
      locationSettings: const LocationSettings(
        accuracy: LocationAccuracy.bestForNavigation,
        // 4 m keeps the arrow lively while walking without burning battery.
        distanceFilter: 4,
      ),
    ).listen(
      (p) => _ingest(
        LatLng(p.latitude, p.longitude),
        p.heading,
        p.speed,
      ),
      onError: (_) {},
    );
  }

  Future<void> setMode(TravelMode m) async {
    if (m == mode) return;
    mode = m;
    await _reroute(force: true);
  }

  void _ingest(LatLng pos, double? gpsHeading, double? speed) {
    _me = pos;
    _speedMps = (speed ?? 0).abs();

    final prog = _computeProgress(pos);
    _progress = prog;

    // --- heading -----------------------------------------------------------
    // Below walking pace the GPS compass is noise, so follow the route's own
    // direction instead. Above it, trust the device.
    double target;
    if (_speedMps > 0.7 && gpsHeading != null && gpsHeading >= 0) {
      target = gpsHeading;
    } else if (prog != null) {
      target = prog.courseBearing;
    } else {
      target = _bearing;
    }
    // Low-pass filter: 35 % of the way to the new angle each fix.
    _bearing = (NavGeo.smoothAngle(_bearing, target, 0.35) + 360) % 360;

    // --- arrival -----------------------------------------------------------
    if (!_arrived &&
        NavGeo.distance(pos, destination) <= arriveThresholdMeters) {
      _arrived = true;
      notifyListeners();
      return;
    }

    // --- off-route detection ----------------------------------------------
    if (prog != null && prog.offRouteMeters > offRouteThresholdMeters) {
      _offRouteStrikes++;
      if (_offRouteStrikes >= 2) _reroute();
    } else {
      _offRouteStrikes = 0;
    }

    notifyListeners();
  }

  RouteProgress? _computeProgress(LatLng pos) {
    final pts = _route.points;
    if (pts.length < 2) return null;

    var bestDist = double.infinity;
    var bestIdx = 0;
    var bestT = 0.0;

    for (var i = 0; i < pts.length - 1; i++) {
      final (d, t) = NavGeo.distanceToSegment(pos, pts[i], pts[i + 1]);
      if (d < bestDist) {
        bestDist = d;
        bestIdx = i;
        bestT = t;
      }
    }

    final snapped = NavGeo.lerp(pts[bestIdx], pts[bestIdx + 1], bestT);

    // Remaining = rest of the current segment + every segment after it.
    var remaining = NavGeo.distance(snapped, pts[bestIdx + 1]);
    for (var i = bestIdx + 1; i < pts.length - 1; i++) {
      remaining += NavGeo.distance(pts[i], pts[i + 1]);
    }

    return RouteProgress(
      snapped: snapped,
      segmentIndex: bestIdx,
      offRouteMeters: bestDist,
      remainingMeters: remaining,
      courseBearing: NavGeo.bearing(pts[bestIdx], pts[bestIdx + 1]),
    );
  }

  Future<void> _reroute({bool force = false}) async {
    if (_rerouting) return;

    // Never hammer the free OSRM demo server.
    final since = DateTime.now().difference(_lastReroute);
    if (!force && since < const Duration(seconds: 8)) return;

    final from = _me;
    if (from == null) return;

    _rerouting = true;
    _lastReroute = DateTime.now();
    notifyListeners();

    RouteResult? fresh;
    if (isOnline()) {
      fresh = await RoutingService.route([from, destination], mode: mode);
    }
    fresh ??= RoutingService.straightLine(from, destination, mode: mode);

    _route = fresh;
    _offRouteStrikes = 0;
    _rerouteCount++;
    _rerouting = false;
    _progress = _computeProgress(from);
    notifyListeners();
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }
}

import '../../data/models/place.dart';
import 'myanmar_text.dart';

/// ---------------------------------------------------------------------------
/// Keyword search + ranking.
///
/// Goals:
///   * one letter is enough to get useful suggestions
///   * searches EVERY field: name (both languages), tags, category, address,
///     description, even the id
///   * multiple words behave like AND ("shan noodle", "lake cafe")
///   * results come back RANKED, so the best match is first instead of
///     whatever order places.json happens to be in
/// ---------------------------------------------------------------------------
class SearchService {
  SearchService._();

  // --- Myanmar script normalisation ----------------------------------------
  //
  // All of the hard work lives in MyanmarText. Briefly:
  //   * Zawgyi text gets converted to Unicode
  //   * left-rendering vowels (ေ ြ) typed before their consonant get moved
  //     back where Unicode stores them
  // Without this, "ဈေး" typed on a phone never matches "ဈေး" in places.json
  // even though the two are pixel-identical on screen.

  /// Consonant skeleton, used for the relaxed fallback pass.
  static String skeleton(String s) => MyanmarText.skeleton(s);

  /// Lowercase, repair Myanmar encoding + ordering, collapse whitespace.
  static String normalize(String s) => MyanmarText.canonical(s.toLowerCase())
      .replaceAll(RegExp(r'\s+'), ' ')
      .trim();

  static List<String> terms(String query) => normalize(query)
      .split(' ')
      .where((w) => w.isNotEmpty)
      .toList(growable: false);

  /// How well a single haystack matches a single term.
  /// 0 = no match. Higher = better.
  static int _fieldScore(String haystack, String term, int weight) {
    if (haystack.isEmpty) return 0;
    final h = normalize(haystack);

    if (h == term) return weight * 4; // exact field
    if (h.startsWith(term)) return weight * 3; // "nau" -> "Naung Tung Lake"

    // Start of any word: "tung" -> "Naung Tung Lake"
    for (final w in h.split(' ')) {
      if (w.startsWith(term)) return weight * 2;
    }

    if (h.contains(term)) return weight; // anywhere
    return 0;
  }

  /// Total score for one place against one term, across all its fields.
  static int _termScore(Place p, String term, String lang, String categoryLabel) {
    var best = 0;

    void take(int v) {
      if (v > best) best = v;
    }

    // Name in the language on screen matters most.
    take(_fieldScore(p.name(lang), term, 100));
    take(_fieldScore(p.name.en, term, 80));
    take(_fieldScore(p.name.my, term, 80));

    for (final tag in p.tags) {
      take(_fieldScore(tag, term, 60));
    }

    take(_fieldScore(categoryLabel, term, 45));
    take(_fieldScore(p.category.id, term, 40));
    take(_fieldScore(p.address(lang), term, 25));
    take(_fieldScore(p.address.en, term, 20));
    take(_fieldScore(p.description(lang), term, 10));
    take(_fieldScore(p.description.en, term, 8));
    take(_fieldScore(p.id.replaceAll('-', ' '), term, 30));

    return best;
  }

  /// Rank [all] against [query]. Empty query returns an empty list — the
  /// caller decides what to show when nothing has been typed.
  ///
  /// [categoryLabelOf] lets the caller pass the localized category name
  /// ("Pagodas" / "ဘုရားများ") so typing a category also works.
  static List<Place> rank(
    List<Place> all,
    String query, {
    required String lang,
    required String Function(Place) categoryLabelOf,
    int limit = 50,
  }) {
    final ts = terms(query);
    if (ts.isEmpty) return const [];

    final scored = <(Place, int)>[];

    for (final p in all) {
      final label = categoryLabelOf(p);
      var total = 0;
      var matchedAll = true;

      for (final term in ts) {
        final s = _termScore(p, term, lang, label);
        if (s == 0) {
          matchedAll = false;
          break;
        }
        total += s;
      }

      if (!matchedAll) continue;

      // Nudge well-rated places up when scores tie.
      total = total * 10 + ((p.rating ?? 0) * 2).round();
      scored.add((p, total));
    }

    scored.sort((a, b) => b.$2.compareTo(a.$2));

    if (scored.isNotEmpty) {
      return scored.take(limit).map((e) => e.$1).toList();
    }

    // Nothing matched exactly. Retry on the consonant skeleton so a missing
    // or extra tone mark does not produce an empty screen.
    return _relaxed(all, ts, lang, categoryLabelOf, limit);
  }

  static List<Place> _relaxed(
    List<Place> all,
    List<String> ts,
    String lang,
    String Function(Place) categoryLabelOf,
    int limit,
  ) {
    final skeletons = ts.map(skeleton).where((e) => e.isNotEmpty).toList();
    if (skeletons.isEmpty) return const [];

    final out = <Place>[];
    for (final p in all) {
      final hay = skeleton(normalize([
        p.name(lang),
        p.name.en,
        p.name.my,
        categoryLabelOf(p),
        p.tags.join(' '),
        p.address(lang),
      ].join(' ')));

      if (skeletons.every(hay.contains)) out.add(p);
      if (out.length >= limit) break;
    }
    return out;
  }

  /// Index of the first term inside [text], for highlighting the matched
  /// part in the suggestion list. Returns (start, end) or null.
  static (int, int)? highlightRange(String text, String query) {
    final ts = terms(query);
    if (ts.isEmpty) return null;
    final h = normalize(text);

    for (final term in ts) {
      final i = h.indexOf(term);
      // The normalized string can be shorter than the original; only use the
      // range when the lengths still line up, otherwise skip highlighting
      // rather than slice the wrong characters.
      if (i >= 0 && h.length == text.length) return (i, i + term.length);
    }
    return null;
  }
}
