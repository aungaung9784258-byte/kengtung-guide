/// This file adds its own provider so lib/core/providers/app_providers.dart

final aboutContentProvider = FutureProvider<AboutContent>((ref) async {
  final raw = await rootBundle.loadString('assets/data/about_kengtung.json');
  return AboutContent.fromJson(json.decode(raw) as Map<String, dynamic>);
});

class AboutKengtungScreen extends ConsumerWidget {
  const AboutKengtungScreen({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final async = ref.watch(aboutContentProvider);
    final lang = ref.watch(langProvider);

    return Scaffold(
      body: async.when(
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (e, _) => _ErrorView(
          message: lang == 'my'
              ? 'အကြောင်းအရာ ဖတ်၍ မရပါ။'
              : 'Could not load this page.',
        ),
        data: (content) => _AboutBody(content: content, lang: lang),
      ),
    );
  }
}

class _Section extends StatelessWidget {
  const _Section({required this.section, required this.lang});

  final AboutSection section;
  final String lang;

  bool get _my => lang == 'my';
  String _t(L10nText v) => _my ? v.my : v.en;

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;

    final bodyStyle = TextStyle(
      fontSize: _my ? 13.5 : 14,
      height: _my ? 1.95 : 1.68,
      color: cs.onSurface.withValues(alpha: 0.88),
    );

    // Everything is padded by hand instead of wrapping the whole column in a
    // Padding, so the big figures can run edge to edge without needing an
    // OverflowBox (which breaks layout inside a sliver).
    Widget pad(Widget child) => Padding(
          padding: const EdgeInsets.symmetric(horizontal: 22),
          child: child,
        );

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        const SizedBox(height: 28),
        pad(_SectionHeading(number: section.number, title: _t(section.title))),
        const SizedBox(height: 12),

        for (final p in section.body) ...[
          pad(Text(_t(p), style: bodyStyle)),
          const SizedBox(height: 12),
        ],

        if (section.type == 'stats')
          pad(_Stats(section: section, lang: lang)),
        if (section.type == 'timeline')
          pad(_Timeline(entries: section.timeline, lang: lang)),
        if (section.names.isNotEmpty) pad(_NameGrid(names: section.names)),
        if (section.type == 'pills') pad(_Pills(pills: section.pills)),
        if (section.type == 'landmarks')
          pad(_Landmarks(landmarks: section.landmarks, lang: lang)),
        if (section.type == 'rows') pad(_Rows(rows: section.rows, lang: lang)),

        if (section.note != null) ...[
          const SizedBox(height: 10),
          pad(_Note(text: _t(section.note!))),
        ],

        if (section.cta != null) ...[
          const SizedBox(height: 14),
          pad(_Cta(label: _t(section.cta!.label), route: section.cta!.route)),
        ],

        if (section.pair.length == 2) ...[
          const SizedBox(height: 16),
          pad(Row(
            children: [
              for (var i = 0; i < 2; i++) ...[
                if (i == 1) const SizedBox(width: 10),
                Expanded(
                  child: FrostedPhoto(
                    image: section.pair[i].image,
                    height: 210,
                    parallax: section.pair[i].parallax,
                    caption: _t(section.pair[i].caption),
                    bandHeight: 96,
                    borderRadius: BorderRadius.circular(18),
                    captionStyle: const TextStyle(
                      color: Colors.white,
                      fontSize: 11,
                      fontWeight: FontWeight.w700,
                    ),
                  ),
                ),
              ],
            ],
          )),
        ],

        if (section.disclaimer != null) ...[
          const SizedBox(height: 8),
          pad(Text(
            _t(section.disclaimer!),
            style: TextStyle(
              fontSize: 10.5,
              height: 1.6,
              fontStyle: FontStyle.italic,
              color: cs.onSurfaceVariant,
            ),
          )),
        ],

        // Full-bleed figure: no horizontal padding at all.
        if (section.figure != null) ...[
          const SizedBox(height: 22),
          FrostedPhoto(
            image: section.figure!.image,
            height: 330,
            parallax: section.figure!.parallax,
            caption: _t(section.figure!.caption),
            disclaimer: section.figure!.disclaimer == null
                ? null
                : _t(section.figure!.disclaimer!),
            bandHeight: 150,
          ),
        ],
      ],
    );
  }
}

import 'dart:ui' as ui;

import 'package:flutter/material.dart';

import '../../../app/theme/app_motion.dart';

/// ---------------------------------------------------------------------------
/// FrostedPhoto
///
/// A photo with two things on top of it:
///   1. a parallax drift, driven by the scroll position of whatever scrollable
///      it happens to live inside, and
///   2. a frosted band along the bottom, so the caption stays readable no
///      matter how bright or busy the photo is.
///
/// The band is a *progressive* blur: three stacked BackdropFilters of
/// increasing sigma. A single blurred rectangle leaves a hard edge where the
/// blur stops; stacking them makes the transition invisible.
///
/// Reusable anywhere a caption sits on a photo — place detail, village cards.
/// ---------------------------------------------------------------------------
class FrostedPhoto extends StatefulWidget {
  const FrostedPhoto({
    super.key,
    required this.image,
    required this.height,
    this.parallax = 0.16,
    this.caption,
    this.disclaimer,
    this.borderRadius = BorderRadius.zero,
    this.bandHeight,
    this.captionStyle,
    this.semanticLabel,
  });

  final String image;
  final double height;

  /// 0.0 = pinned, 0.30 = drifts a lot. Comes from the JSON per figure.
  final double parallax;
  final String? caption;

  /// Small italic line under the caption (e.g. "generated by AI").
  final String? disclaimer;
  final BorderRadius borderRadius;

  /// Defaults to 45% of [height].
  final double? bandHeight;
  final TextStyle? captionStyle;
  final String? semanticLabel;

  @override
  State<FrostedPhoto> createState() => _FrostedPhotoState();
}

class _FrostedPhotoState extends State<FrostedPhoto> {
  ScrollPosition? _position;
  double _shift = 0;

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    _position?.removeListener(_onScroll);
    _position = Scrollable.maybeOf(context)?.position;
    _position?.addListener(_onScroll);
    WidgetsBinding.instance.addPostFrameCallback((_) => _onScroll());
  }

  @override
  void dispose() {
    _position?.removeListener(_onScroll);
    super.dispose();
  }

  void _onScroll() {
    if (!mounted) return;
    final box = context.findRenderObject() as RenderBox?;
    final scrollBox =
        Scrollable.maybeOf(context)?.context.findRenderObject() as RenderBox?;
    if (box == null || scrollBox == null || !box.hasSize) return;

    // How far this photo's centre is from the centre of the viewport.
    final selfCentre =
        box.localToGlobal(box.size.center(Offset.zero), ancestor: scrollBox).dy;
    final viewportCentre = scrollBox.size.height / 2;
    final next = (selfCentre - viewportCentre) * -widget.parallax;

    if ((next - _shift).abs() > 0.3) setState(() => _shift = next);
  }

  @override
  Widget build(BuildContext context) {
    final band = widget.bandHeight ?? widget.height * 0.45;
    // The image is taller than its frame so there is room to drift.
    final overscan = widget.height * widget.parallax * 2.2;

    return ClipRRect(

  Widget _blurBand({required double height, required double sigma}) {
    return Align(
      alignment: Alignment.bottomCenter,
      child: SizedBox(
        height: height,
        width: double.infinity,
        child: ClipRect(
          child: BackdropFilter(
            filter: ui.ImageFilter.blur(sigmaX: sigma, sigmaY: sigma),
            child: const SizedBox.expand(),
          ),
        ),
      ),
    );
